   258→   258→// Fish catch logic
   259→   259→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   260→   260→  const zone = WATER_ZONES[zoneId];
   261→   261→  const rod = ROD_TIERS[rodTier];
   262→   262→
   263→   263→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   264→   264→  const totalWeight = speciesList.reduce((sum, sp) => {
   265→   265→    let w = sp.dropWeight;
   266→   266→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   267→   267→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   268→   268→    return sum + w;
   269→   269→  }, 0);
   270→   270→
   271→   271→  let roll = Math.random() * totalWeight;
   272→   272→  for (const sp of speciesList) {
   273→   273→    let w = sp.dropWeight;
   274→   274→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   275→   275→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   276→   276→    roll -= w;
   277→   277→    if (roll <= 0) {
   278→   278→      const codexKey = `fish_${sp.id}`;
   279→   279→      const isNew = !fishCodex[codexKey];
   280→   280→      return { speciesId: sp.id, isNew };
   281→   281→    }
   282→   282→  }
   283→   283→  return null;
   284→   284→}

   286→   286→// Smuggler loot roll
   287→   287→function rollSmugglerLoot(): { itemId: string; qty: number } {
   288→   288→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   289→   289→  let roll = Math.random() * totalWeight;
   290→   290→  for (const entry of SMUGGLER_LOOT) {
   291→   291→    roll -= entry.weight;
   292→   292→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   293→   293→  }
   294→   294→  return { itemId: 'gems', qty: 1 };
   295→   295→}

   303→   303→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   304→   304→  let newXp = xp;
   305→   305→  let newLevel = level;
   306→   306→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   307→   307→    newXp -= xpForLevel(newLevel);
   308→   308→    newLevel++;
   309→   309→  }
   310→   310→  return { xp: newXp, level: newLevel };
   311→   311→}
