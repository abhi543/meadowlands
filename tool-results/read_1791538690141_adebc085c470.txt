   284→function generateOrders(count: number): OrderState[] {
   285→  const orders: OrderState[] = [];
   286→  const pool = [...ORDER_TEMPLATES];
   287→  for (let i = 0; i < count; i++) {
   288→    const template = pool[Math.floor(Math.random() * pool.length)];
   289→    orders.push({
   290→      id: `order_${Date.now()}_${i}`,
   291→      ...template,
   292→      trashedAt: null,
   293→    });
   294→  }
   295→  return orders;

   342→// Fish catch logic
   343→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   344→  const zone = WATER_ZONES[zoneId];
   345→  const rod = ROD_TIERS[rodTier];
   346→
   347→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   348→  const totalWeight = speciesList.reduce((sum, sp) => {
   349→    let w = sp.dropWeight;
   350→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   351→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   352→    return sum + w;
   353→  }, 0);
   354→
   355→  let roll = Math.random() * totalWeight;
   356→  for (const sp of speciesList) {
   357→    let w = sp.dropWeight;
   358→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   359→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   360→    roll -= w;
   361→    if (roll <= 0) {
   362→      const codexKey = `fish_${sp.id}`;
   363→      const isNew = !fishCodex[codexKey];
   364→      return { speciesId: sp.id, isNew };
   365→    }
   366→  }
   367→  return null;
   368→}

   370→// Smuggler loot roll
   371→function rollSmugglerLoot(): { itemId: string; qty: number } {
   372→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   373→  let roll = Math.random() * totalWeight;
   374→  for (const entry of SMUGGLER_LOOT) {
   375→    roll -= entry.weight;
   376→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   377→  }
   378→  return { itemId: 'gems', qty: 1 };
   379→}

   387→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   388→  let newXp = xp;
   389→  let newLevel = level;
   390→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   391→    newXp -= xpForLevel(newLevel);
   392→    newLevel++;
   393→  }
   394→  return { xp: newXp, level: newLevel };
   395→}
