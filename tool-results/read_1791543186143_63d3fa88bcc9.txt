   303→function generateOrders(count: number): OrderState[] {
   304→  const orders: OrderState[] = [];
   305→  const pool = [...ORDER_TEMPLATES];
   306→  for (let i = 0; i < count; i++) {
   307→    const template = pool[Math.floor(Math.random() * pool.length)];
   308→    orders.push({
   309→      id: `order_${Date.now()}_${i}`,
   310→      ...template,
   311→      trashedAt: null,
   312→    });
   313→  }
   314→  return orders;

   380→// Fish catch logic
   381→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   382→  const zone = WATER_ZONES[zoneId];
   383→  const rod = ROD_TIERS[rodTier];
   384→
   385→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   386→  const totalWeight = speciesList.reduce((sum, sp) => {
   387→    let w = sp.dropWeight;
   388→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   389→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   390→    return sum + w;
   391→  }, 0);
   392→
   393→  let roll = Math.random() * totalWeight;
   394→  for (const sp of speciesList) {
   395→    let w = sp.dropWeight;
   396→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   397→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   398→    roll -= w;
   399→    if (roll <= 0) {
   400→      const codexKey = `fish_${sp.id}`;
   401→      const isNew = !fishCodex[codexKey];
   402→      return { speciesId: sp.id, isNew };
   403→    }
   404→  }
   405→  return null;
   406→}

   408→// Smuggler loot roll
   409→function rollSmugglerLoot(): { itemId: string; qty: number } {
   410→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   411→  let roll = Math.random() * totalWeight;
   412→  for (const entry of SMUGGLER_LOOT) {
   413→    roll -= entry.weight;
   414→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   415→  }
   416→  return { itemId: 'gems', qty: 1 };
   417→}

   425→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   426→  let newXp = xp;
   427→  let newLevel = level;
   428→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   429→    newXp -= xpForLevel(newLevel);
   430→    newLevel++;
   431→  }
   432→  return { xp: newXp, level: newLevel };
   433→}
