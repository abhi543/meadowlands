    29→// Randomize max cycles for a new animal: ±30% around the tier base, clamped to [4, 30]
    30→export function randomMaxCycles(tier: Tier): number {
    31→  const base = TIER_MAX_CYCLES[tier];
    32→  const min = Math.max(4, Math.floor(base * 0.7));
    33→  const max = Math.min(30, Math.ceil(base * 1.3));
    34→  return min + Math.floor(Math.random() * (max - min + 1));
    35→}

   192→// Helper: get the default breedId for a given species (common tier)
   193→export function defaultBreedId(species: AnimalSpecies): string {
   194→  return `${species}_common`;
   195→}

   197→// Helper: get all breed IDs for a given species
   198→export function breedsForSpecies(species: AnimalSpecies): string[] {
   199→  return Object.values(BREED_REGISTRY).filter(b => b.species === species).map(b => b.id);
   200→}
