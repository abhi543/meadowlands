   252→function generateOrders(count: number): OrderState[] {
   253→  const orders: OrderState[] = [];
   254→  const pool = [...ORDER_TEMPLATES];
   255→  for (let i = 0; i < count; i++) {
   256→    const template = pool[Math.floor(Math.random() * pool.length)];
   257→    orders.push({
   258→      id: `order_${Date.now()}_${i}`,
   259→      ...template,
   260→      trashedAt: null,
   261→    });
   262→  }
   263→  return orders;
   264→}

   285→// Fish catch logic
   286→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   287→  const zone = WATER_ZONES[zoneId];
   288→  const rod = ROD_TIERS[rodTier];
   289→
   290→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   291→  const totalWeight = speciesList.reduce((sum, sp) => {
   292→    let w = sp.dropWeight;
   293→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   294→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   295→    return sum + w;
   296→  }, 0);
   297→
   298→  let roll = Math.random() * totalWeight;
   299→  for (const sp of speciesList) {
   300→    let w = sp.dropWeight;
   301→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   302→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   303→    roll -= w;
   304→    if (roll <= 0) {
   305→      const codexKey = `fish_${sp.id}`;
   306→      const isNew = !fishCodex[codexKey];
   307→      return { speciesId: sp.id, isNew };
   308→    }
   309→  }
   310→  return null;
   311→}

   313→// Smuggler loot roll
   314→function rollSmugglerLoot(): { itemId: string; qty: number } {
   315→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   316→  let roll = Math.random() * totalWeight;
   317→  for (const entry of SMUGGLER_LOOT) {
   318→    roll -= entry.weight;
   319→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   320→  }
   321→  return { itemId: 'gems', qty: 1 };
   322→}

   330→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   331→  let newXp = xp;
   332→  let newLevel = level;
   333→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   334→    newXp -= xpForLevel(newLevel);
   335→    newLevel++;
   336→  }
   337→  return { xp: newXp, level: newLevel };
   338→}

   445→        if (cropDef.seedCost > 0 && state.coins < totalCost) return;

   537→          return { ...a, fed: false, fedAt: null, productReady: false, cyclesRemaining: newCyclesRemaining, age: isElder ? 'elder' : a.age };

   548→        if (currentCount >= (state.animalCapacity[species] || def.capacity)) return;
   549→        const costs: Record<AnimalSpecies, number> = { chicken: 50, cow: 150, sheep: 200, pig: 300 };
