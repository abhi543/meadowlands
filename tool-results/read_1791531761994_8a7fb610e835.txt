    29→// Randomize max cycles for a new animal: ±30% around the tier base, clamped to [4, 30]
    30→export function randomMaxCycles(tier: Tier): number {
    31→  const base = TIER_MAX_CYCLES[tier];
    32→  const min = Math.max(4, Math.floor(base * 0.7));
    33→  const max = Math.min(30, Math.ceil(base * 1.3));
    34→  return min + Math.floor(Math.random() * (max - min + 1));
    35→}

   693→// Proportional market price based on cycles already completed

   695→export function livestockSellPriceForAnimal(species: AnimalSpecies, tier: Tier, cyclesRemaining: number, maxCycles: number): number {

   697→  const cyclesCompleted = maxCycles - cyclesRemaining;
   698→  if (cyclesCompleted <= 0) return 0; // haven't produced anything yet
   699→  if (cyclesRemaining <= 0) return fullPrice; // elder — full payout
   700→  return Math.floor(fullPrice * (cyclesCompleted / maxCycles));

   854→/** Cost to upgrade from level L to L+1 */
   855→export function workerUpgradeCost(role: WorkerRole, currentLevel: number): number {
   856→  if (currentLevel >= WORKER_MAX_LEVEL) return 0;
   857→  const cfg = WORKER_REGISTRY[role];
   858→  const linearDamp = 1 + currentLevel / 12;
   859→  return Math.floor(cfg.baseCost * Math.pow(cfg.exponent, currentLevel - 1) * linearDamp);
   860→}
   861→
   862→/** Speed multiplier: asymptotically approaches 0.25× at level 120 */
   863→export function workerSpeedMultiplier(level: number): number {
   864→  const progressRatio = Math.max(0, Math.min(1, (WORKER_MAX_LEVEL - level) / (WORKER_MAX_LEVEL - 1)));
   865→  return 0.25 + 0.75 * Math.pow(progressRatio, 1.15);
   866→}
   867→
   868→/** Overdrive bonus: additional -20% at level 120 */
   869→export function workerOverdrive(level: number): number {
   870→  return level >= WORKER_MAX_LEVEL ? 0.8 : 1.0;
   871→}
   872→
   873→/** Effective interval in ms for a worker at given level */
   874→export function workerEffectiveInterval(role: WorkerRole, level: number): number {
   875→  const cfg = WORKER_REGISTRY[role];
   876→  return Math.floor(cfg.baseIntervalMs * workerSpeedMultiplier(level) * workerOverdrive(level));
   877→}

   940→// --- Bulk Upgrade Engine ---
   941→
   942→export type BulkMode = 'x1' | 'x10' | 'MAX';
   943→
   944→export interface BulkUpgradeResult {
   945→  levelsToBuy: number;
   946→  totalCost: number;
   947→}
   948→
   949→export function calculateBulkUpgrade(role: WorkerRole, currentLevel: number, mode: BulkMode, currentCoins: number): BulkUpgradeResult {
   950→  if (currentLevel >= WORKER_MAX_LEVEL) return { levelsToBuy: 0, totalCost: 0 };
   951→
   952→  if (mode === 'x1') {
   953→    const cost = workerUpgradeCost(role, currentLevel);
   954→    return currentCoins >= cost ? { levelsToBuy: 1, totalCost: cost } : { levelsToBuy: 0, totalCost: cost };
   955→  }
   956→
   957→  if (mode === 'x10') {
   958→    const maxLevels = Math.min(10, WORKER_MAX_LEVEL - currentLevel);
   959→    let totalCost = 0;
   960→    let affordable = 0;
   961→    for (let i = 0; i < maxLevels; i++) {
   962→      const cost = workerUpgradeCost(role, currentLevel + i);
   963→      if (totalCost + cost > currentCoins) break;
   964→      totalCost += cost;
   965→      affordable++;
   966→    }
   967→    return { levelsToBuy: affordable, totalCost };
   968→  }
   969→
   970→  // MAX mode
   971→  let levelsToBuy = 0;
   972→  let totalCost = 0;
   973→  while (currentLevel + levelsToBuy < WORKER_MAX_LEVEL) {
   974→    const nextCost = workerUpgradeCost(role, currentLevel + levelsToBuy);
   975→    if (totalCost + nextCost > currentCoins) break;
   976→    totalCost += nextCost;
   977→    levelsToBuy++;
   978→  }
   979→  return { levelsToBuy, totalCost };
   980→}
   981→
   982→/** Get the total payout for a worker in auto-sell mode (per cycle) */
   983→export function workerAutoSellPayout(role: WorkerRole, level: number): number {
   984→  const cfg = WORKER_REGISTRY[role];
   985→  return Math.round(cfg.autoSellPayout * workerAutoSellBonus(level));
   986→}
