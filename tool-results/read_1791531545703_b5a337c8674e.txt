   281→function generateOrders(count: number): OrderState[] {
   282→  const orders: OrderState[] = [];
   283→  const pool = [...ORDER_TEMPLATES];
   284→  for (let i = 0; i < count; i++) {
   285→    const template = pool[Math.floor(Math.random() * pool.length)];
   286→    orders.push({
   287→      id: `order_${Date.now()}_${i}`,
   288→      ...template,
   289→      trashedAt: null,
   290→    });
   291→  }
   292→  return orders;
   293→}

   339→// Fish catch logic
   340→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   341→  const zone = WATER_ZONES[zoneId];
   342→  const rod = ROD_TIERS[rodTier];
   343→
   344→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   345→  const totalWeight = speciesList.reduce((sum, sp) => {
   346→    let w = sp.dropWeight;
   347→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   348→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   349→    return sum + w;
   350→  }, 0);
   351→
   352→  let roll = Math.random() * totalWeight;
   353→  for (const sp of speciesList) {
   354→    let w = sp.dropWeight;
   355→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   356→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   357→    roll -= w;
   358→    if (roll <= 0) {
   359→      const codexKey = `fish_${sp.id}`;
   360→      const isNew = !fishCodex[codexKey];
   361→      return { speciesId: sp.id, isNew };
   362→    }
   363→  }
   364→  return null;
   365→}

   367→// Smuggler loot roll
   368→function rollSmugglerLoot(): { itemId: string; qty: number } {
   369→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   370→  let roll = Math.random() * totalWeight;
   371→  for (const entry of SMUGGLER_LOOT) {
   372→    roll -= entry.weight;
   373→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   374→  }
   375→  return { itemId: 'gems', qty: 1 };
   376→}

   384→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   385→  let newXp = xp;
   386→  let newLevel = level;
   387→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   388→    newXp -= xpForLevel(newLevel);
   389→    newLevel++;
   390→  }
   391→  return { xp: newXp, level: newLevel };
   392→}
