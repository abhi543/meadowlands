    29→// Randomize max cycles for a new animal: ±30% around the tier base, clamped to [4, 30]
    30→export function randomMaxCycles(tier: Tier): number {
    31→  const base = TIER_MAX_CYCLES[tier];
    32→  const min = Math.max(4, Math.floor(base * 0.7));
    33→  const max = Math.min(30, Math.ceil(base * 1.3));
    34→  return min + Math.floor(Math.random() * (max - min + 1));
    35→}

   665→// --- Livestock Market ---
   666→export const ANIMAL_BUY_PRICES: Record<AnimalSpecies, number> = { chicken: 50, cow: 150, sheep: 200, pig: 300 };
   667→
   668→// Market sale prices (common tier base) — derived from the economic formula:
   669→//   Market Sale Payout = Purchase Cost + (Σ Commodity Value Generated × 1.25)
   670→// Chicken: 50 + (150 × 1.25) = 237.5 → 250
   671→// Cow:    150 + (192 × 1.25) = 390   → 750  (higher tier investment reward)
   672→// Sheep:  200 + (200 × 1.25) = 450   → 1200 (premium for patience)
   673→// Pig:    300 + (240 × 1.25) = 600   → 2200 (jackpot payday animal)
   674→export const ANIMAL_MARKET_PRICES: Record<AnimalSpecies, number> = {
   675→  chicken: 250,
   676→  cow: 750,
   677→  sheep: 1200,
   678→  pig: 2200,
   679→};
   680→
   681→export const TIER_MULTIPLIER: Record<Tier, number> = {
   682→  common: 1.0,
   683→  uncommon: 1.5,
   684→  rare: 2.5,
   685→  legendary: 5.0,
   686→};
   687→
   688→// Full market price for an elder animal (completed all productive cycles)
   689→export function livestockSellPrice(species: AnimalSpecies, tier: Tier): number {
   690→  return Math.floor(ANIMAL_MARKET_PRICES[species] * TIER_MULTIPLIER[tier]);
   691→}
   692→
   693→// Proportional market price based on cycles already completed
   694→// Elder animals (cyclesRemaining=0) get full price; younger animals get a fraction
   695→export function livestockSellPriceForAnimal(species: AnimalSpecies, tier: Tier, cyclesRemaining: number, maxCycles: number): number {
   696→  const fullPrice = livestockSellPrice(species, tier);
   697→  const cyclesCompleted = maxCycles - cyclesRemaining;
   698→  if (cyclesCompleted <= 0) return 0; // haven't produced anything yet
   699→  if (cyclesRemaining <= 0) return fullPrice; // elder — full payout
   700→  return Math.floor(fullPrice * (cyclesCompleted / maxCycles));
   701→}
   702→
   703→export function livestockBuyPrice(species: AnimalSpecies, tier: Tier): number {
   704→  return Math.floor(ANIMAL_BUY_PRICES[species] * TIER_MULTIPLIER[tier]);
   705→}
