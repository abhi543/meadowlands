    24→function formatDuration(ms: number): string {
    25→  const totalSec = Math.max(0, Math.ceil(ms / 1000));
    26→  if (totalSec < 60) return `${totalSec}s`;
    27→  const min = Math.floor(totalSec / 60);
    28→  const sec = totalSec % 60;
    29→  if (min < 60) return `${min}m ${sec}s`;
    30→  const hr = Math.floor(min / 60);
    31→  const rm = min % 60;
    32→  return `${hr}h ${rm}m`;
    33→}
    34→
    35→function formatCoin(n: number): string {
    36→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    37→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    38→  return n.toLocaleString();
    39→}
