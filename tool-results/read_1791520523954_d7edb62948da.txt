   224→function generateOrders(count: number): OrderState[] {
   225→  const orders: OrderState[] = [];
   226→  const pool = [...ORDER_TEMPLATES];
   227→  for (let i = 0; i < count; i++) {
   228→    const template = pool[Math.floor(Math.random() * pool.length)];
   229→    orders.push({
   230→      id: `order_${Date.now()}_${i}`,
   231→      ...template,
   232→      trashedAt: null,
   233→    });
   234→  }
   235→  return orders;
   236→}

   257→// Fish catch logic
   258→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   259→  const zone = WATER_ZONES[zoneId];
   260→  const rod = ROD_TIERS[rodTier];
   261→
   262→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   263→  const totalWeight = speciesList.reduce((sum, sp) => {
   264→    let w = sp.dropWeight;
   265→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   266→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   267→    return sum + w;
   268→  }, 0);
   269→
   270→  let roll = Math.random() * totalWeight;
   271→  for (const sp of speciesList) {
   272→    let w = sp.dropWeight;
   273→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   274→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   275→    roll -= w;
   276→    if (roll <= 0) {
   277→      const codexKey = `fish_${sp.id}`;
   278→      const isNew = !fishCodex[codexKey];
   279→      return { speciesId: sp.id, isNew };
   280→    }
   281→  }
   282→  return null;
   283→}

   285→// Smuggler loot roll
   286→function rollSmugglerLoot(): { itemId: string; qty: number } {
   287→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   288→  let roll = Math.random() * totalWeight;
   289→  for (const entry of SMUGGLER_LOOT) {
   290→    roll -= entry.weight;
   291→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   292→  }
   293→  return { itemId: 'gems', qty: 1 };
   294→}

   302→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   303→  let newXp = xp;
   304→  let newLevel = level;
   305→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   306→    newXp -= xpForLevel(newLevel);
   307→    newLevel++;
   308→  }
   309→  return { xp: newXp, level: newLevel };
   310→}
