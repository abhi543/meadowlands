   848→
   849→// Market sale prices (common tier base) — derived from the economic formula:
   850→//   Market Sale Payout = Purchase Cost + (Σ Commodity Value Generated × 1.25)
   851→// Chicken: 50 + (150 × 1.25) = 237.5 → 250
   852→// Cow:    150 + (192 × 1.25) = 390   → 750  (higher tier investment reward)
   853→// Sheep:  200 + (200 × 1.25) = 450   → 1200 (premium for patience)
   854→// Pig:    300 + (240 × 1.25) = 600   → 2200 (jackpot payday animal)
   855→export const ANIMAL_MARKET_PRICES: Record<AnimalSpecies, number> = {
   856→  chicken: 250,
   857→  cow: 750,
   858→  sheep: 1200,
   859→  pig: 2200,

   862→export const TIER_MULTIPLIER: Record<Tier, number> = {
   863→  common: 1.0,
   864→  uncommon: 1.5,
   865→  rare: 2.5,
   866→  legendary: 5.0,
   867→};
   868→
   869→// Full market price for an elder animal (completed all productive cycles)
   870→export function livestockSellPrice(species: AnimalSpecies, tier: Tier): number {
   871→  return Math.floor(ANIMAL_MARKET_PRICES[species] * TIER_MULTIPLIER[tier]);
   872→}
   873→
   874→// Proportional market price based on cycles already completed
   875→// Elder animals (cyclesRemaining=0) get full price; younger animals get a fraction
   876→export function livestockSellPriceForAnimal(species: AnimalSpecies, tier: Tier, cyclesRemaining: number, maxCycles: number): number {
   877→  const fullPrice = livestockSellPrice(species, tier);
   878→  const cyclesCompleted = maxCycles - cyclesRemaining;
   879→  if (cyclesCompleted <= 0) return 0; // haven't produced anything yet
   880→  if (cyclesRemaining <= 0) return fullPrice; // elder — full payout
   881→  return Math.floor(fullPrice * (cyclesCompleted / maxCycles));
   882→}
   883→
   884→export function livestockBuyPrice(species: AnimalSpecies, tier: Tier): number {
   885→  return Math.floor(ANIMAL_BUY_PRICES[species] * TIER_MULTIPLIER[tier]);
   886→}

  1031→// --- 120-Level Progression Math ---

  1033→export const WORKER_MAX_LEVEL = 120;
  1034→
  1035→/** Cost to upgrade from level L to L+1 */
  1036→export function workerUpgradeCost(role: WorkerRole, currentLevel: number): number {
  1037→  if (currentLevel >= WORKER_MAX_LEVEL) return 0;
  1038→  const cfg = WORKER_REGISTRY[role];
  1039→  const linearDamp = 1 + currentLevel / 12;
  1040→  return Math.floor(cfg.baseCost * Math.pow(cfg.exponent, currentLevel - 1) * linearDamp);
  1041→}
  1042→
  1043→/** Speed multiplier: asymptotically approaches 0.25× at level 120 */
  1044→export function workerSpeedMultiplier(level: number): number {
  1045→  const progressRatio = Math.max(0, Math.min(1, (WORKER_MAX_LEVEL - level) / (WORKER_MAX_LEVEL - 1)));
  1046→  return 0.25 + 0.75 * Math.pow(progressRatio, 1.15);
  1047→}
  1048→
  1049→/** Overdrive bonus: additional -20% at level 120 */
  1050→export function workerOverdrive(level: number): number {
  1051→  return level >= WORKER_MAX_LEVEL ? 0.8 : 1.0;
  1052→}
  1053→
  1054→/** Effective interval in ms for a worker at given level */
  1055→export function workerEffectiveInterval(role: WorkerRole, level: number): number {
  1056→  const cfg = WORKER_REGISTRY[role];
  1057→  return Math.floor(cfg.baseIntervalMs * workerSpeedMultiplier(level) * workerOverdrive(level));
  1058→}
  1059→
  1060→/** Double proc chance: 0% at L1, ~60% at L120 */
  1061→export function workerDoubleProcChance(level: number): number {
  1062→  return Math.min(0.60, (level - 1) / 200);
  1063→}

  1121→// --- Bulk Upgrade Engine ---
  1122→
  1123→export type BulkMode = 'x1' | 'x10' | 'MAX';
  1124→
  1125→export interface BulkUpgradeResult {
  1126→  levelsToBuy: number;
  1127→  totalCost: number;
  1128→}
  1129→
  1130→export function calculateBulkUpgrade(role: WorkerRole, currentLevel: number, mode: BulkMode, currentCoins: number): BulkUpgradeResult {
  1131→  if (currentLevel >= WORKER_MAX_LEVEL) return { levelsToBuy: 0, totalCost: 0 };
  1132→
  1133→  if (mode === 'x1') {
  1134→    const cost = workerUpgradeCost(role, currentLevel);
  1135→    return currentCoins >= cost ? { levelsToBuy: 1, totalCost: cost } : { levelsToBuy: 0, totalCost: cost };
  1136→  }
  1137→
  1138→  if (mode === 'x10') {
  1139→    const maxLevels = Math.min(10, WORKER_MAX_LEVEL - currentLevel);
  1140→    let totalCost = 0;
  1141→    let affordable = 0;
  1142→    for (let i = 0; i < maxLevels; i++) {
  1143→      const cost = workerUpgradeCost(role, currentLevel + i);
  1144→      if (totalCost + cost > currentCoins) break;
  1145→      totalCost += cost;
  1146→      affordable++;
  1147→    }
  1148→    return { levelsToBuy: affordable, totalCost };
  1149→  }
  1150→
  1151→  // MAX mode
  1152→  let levelsToBuy = 0;
  1153→  let totalCost = 0;
  1154→  while (currentLevel + levelsToBuy < WORKER_MAX_LEVEL) {
  1155→    const nextCost = workerUpgradeCost(role, currentLevel + levelsToBuy);
  1156→    if (totalCost + nextCost > currentCoins) break;
  1157→    totalCost += nextCost;
  1158→    levelsToBuy++;
  1159→  }
  1160→  return { levelsToBuy, totalCost };
  1161→}

  1163→/** Get the total payout for a worker in auto-sell mode (per cycle) */
  1164→export function workerAutoSellPayout(role: WorkerRole, level: number): number {
  1165→  const cfg = WORKER_REGISTRY[role];
  1166→  return Math.round(cfg.autoSellPayout * workerAutoSellBonus(level));
  1167→}
