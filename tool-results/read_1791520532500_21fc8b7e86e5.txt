   257→   257→// Fish catch logic
   258→   258→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   259→   259→  const zone = WATER_ZONES[zoneId];
   260→   260→  const rod = ROD_TIERS[rodTier];
   261→   261→
   262→   262→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   263→   263→  const totalWeight = speciesList.reduce((sum, sp) => {
   264→   264→    let w = sp.dropWeight;
   265→   265→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   266→   266→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   267→   267→    return sum + w;
   268→   268→  }, 0);
   269→   269→
   270→   270→  let roll = Math.random() * totalWeight;
   271→   271→  for (const sp of speciesList) {
   272→   272→    let w = sp.dropWeight;
   273→   273→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   274→   274→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   275→   275→    roll -= w;
   276→   276→    if (roll <= 0) {
   277→   277→      const codexKey = `fish_${sp.id}`;
   278→   278→      const isNew = !fishCodex[codexKey];
   279→   279→      return { speciesId: sp.id, isNew };
   280→   280→    }
   281→   281→  }
   282→   282→  return null;
   283→   283→}

   285→   285→// Smuggler loot roll
   286→   286→function rollSmugglerLoot(): { itemId: string; qty: number } {
   287→   287→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   288→   288→  let roll = Math.random() * totalWeight;
   289→   289→  for (const entry of SMUGGLER_LOOT) {
   290→   290→    roll -= entry.weight;
   291→   291→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   292→   292→  }
   293→   293→  return { itemId: 'gems', qty: 1 };
   294→   294→}

   302→   302→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   303→   303→  let newXp = xp;
   304→   304→  let newLevel = level;
   305→   305→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   306→   306→    newXp -= xpForLevel(newLevel);
   307→   307→    newLevel++;
   308→   308→  }
   309→   309→  return { xp: newXp, level: newLevel };
   310→   310→}
