   221→      ...template,
   222→      trashedAt: null,

   247→// Fish catch logic
   248→function rollFishCatch(zoneId: WaterZoneId, rodTier: RodTier, fishCodex: Record<string, boolean>): { speciesId: FishSpeciesId; isNew: boolean } | null {
   249→  const zone = WATER_ZONES[zoneId];
   250→  const rod = ROD_TIERS[rodTier];
   251→
   252→  const speciesList = zone.species.map(id => FISH_SPECIES[id]);
   253→  const totalWeight = speciesList.reduce((sum, sp) => {
   254→    let w = sp.dropWeight;
   255→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   256→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   257→    return sum + w;
   258→  }, 0);
   259→
   260→  let roll = Math.random() * totalWeight;
   261→  for (const sp of speciesList) {
   262→    let w = sp.dropWeight;
   263→    if (sp.tier === 'rare') w *= (1 + rod.rareBoost);
   264→    if (sp.tier === 'legendary') w *= (1 + rod.legendaryBoost);
   265→    roll -= w;
   266→    if (roll <= 0) {
   267→      const codexKey = `fish_${sp.id}`;
   268→      const isNew = !fishCodex[codexKey];
   269→      return { speciesId: sp.id, isNew };
   270→    }
   271→  }
   272→  return null;
   273→}

   275→// Smuggler loot roll
   276→function rollSmugglerLoot(): { itemId: string; qty: number } {
   277→  const totalWeight = SMUGGLER_LOOT.reduce((s, e) => s + e.weight, 0);
   278→  let roll = Math.random() * totalWeight;
   279→  for (const entry of SMUGGLER_LOOT) {
   280→    roll -= entry.weight;
   281→    if (roll <= 0) return { itemId: entry.itemId, qty: entry.qty };
   282→  }
   283→  return { itemId: 'gems', qty: 1 };
   284→}

   292→function levelUpCheck(xp: number, level: number): { xp: number; level: number } {
   293→  let newXp = xp;
   294→  let newLevel = level;
   295→  while (newLevel < MAX_LEVEL && newXp >= xpForLevel(newLevel)) {
   296→    newXp -= xpForLevel(newLevel);
   297→    newLevel++;
   298→  }
   299→  return { xp: newXp, level: newLevel };
   300→}
