    29→// Randomize max cycles for a new animal: ±30% around the tier base, clamped to [4, 30]
    30→export function randomMaxCycles(tier: Tier): number {
    31→  const base = TIER_MAX_CYCLES[tier];
    32→  const min = Math.max(4, Math.floor(base * 0.7));
    33→  const max = Math.min(30, Math.ceil(base * 1.3));
    34→  return min + Math.floor(Math.random() * (max - min + 1));
    35→}

   535→// --- Livestock Market ---
   536→export const ANIMAL_BUY_PRICES: Record<AnimalSpecies, number> = { chicken: 50, cow: 150, sheep: 200, pig: 300 };
   537→
   538→// Market sale prices (common tier base) — derived from the economic formula:
   539→//   Market Sale Payout = Purchase Cost + (Σ Commodity Value Generated × 1.25)
   540→// Chicken: 50 + (150 × 1.25) = 237.5 → 250
   541→// Cow:    150 + (192 × 1.25) = 390   → 750  (higher tier investment reward)
   542→// Sheep:  200 + (200 × 1.25) = 450   → 1200 (premium for patience)
   543→// Pig:    300 + (240 × 1.25) = 600   → 2200 (jackpot payday animal)
   544→export const ANIMAL_MARKET_PRICES: Record<AnimalSpecies, number> = {
   545→  chicken: 250,
   546→  cow: 750,
   547→  sheep: 1200,
   548→  pig: 2200,
   549→};
   550→
   551→export const TIER_MULTIPLIER: Record<Tier, number> = {
   552→  common: 1.0,
   553→  uncommon: 1.5,
   554→  rare: 2.5,
   555→  legendary: 5.0,
   556→};
   557→
   558→// Full market price for an elder animal (completed all productive cycles)
   559→export function livestockSellPrice(species: AnimalSpecies, tier: Tier): number {
   560→  return Math.floor(ANIMAL_MARKET_PRICES[species] * TIER_MULTIPLIER[tier]);
   561→}
   562→
   563→// Proportional market price based on cycles already completed
   564→// Elder animals (cyclesRemaining=0) get full price; younger animals get a fraction
   565→export function livestockSellPriceForAnimal(species: AnimalSpecies, tier: Tier, cyclesRemaining: number, maxCycles: number): number {
   566→  const fullPrice = livestockSellPrice(species, tier);
   567→  const cyclesCompleted = maxCycles - cyclesRemaining;
   568→  if (cyclesCompleted <= 0) return 0; // haven't produced anything yet
   569→  if (cyclesRemaining <= 0) return fullPrice; // elder — full payout
   570→  return Math.floor(fullPrice * (cyclesCompleted / maxCycles));
   571→}
   572→
   573→export function livestockBuyPrice(species: AnimalSpecies, tier: Tier): number {
   574→  return Math.floor(ANIMAL_BUY_PRICES[species] * TIER_MULTIPLIER[tier]);
   575→}
